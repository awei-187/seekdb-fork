cmake_minimum_required(VERSION 3.20)
project(file_sql_native_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SEEKDB_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
add_executable(file_sql_native_tests test_file_scan_utils.cpp
  "${SEEKDB_ROOT}/src/sql/engine/basic/file_scan_utils.cpp"
  "${SEEKDB_ROOT}/src/sql/engine/basic/parquet_file_reader.cpp")
target_compile_definitions(file_sql_native_tests PRIVATE FILE_SQL_STANDALONE_TEST)
target_include_directories(file_sql_native_tests PRIVATE "${SEEKDB_ROOT}/src")
if(MSVC)
  target_compile_options(file_sql_native_tests PRIVATE /utf-8 /EHsc)
endif()
enable_testing()
add_test(NAME file_sql_native COMMAND file_sql_native_tests)
set(FILE_SQL_FIXTURE_DIR "" CACHE PATH "Directory generated by generate_parquet_fixtures.py")
if(FILE_SQL_FIXTURE_DIR)
  add_test(NAME file_sql_parquet_reference COMMAND file_sql_native_tests "${FILE_SQL_FIXTURE_DIR}")
endif()
